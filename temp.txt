def launch_game(game_code, playtime_minutes):
    if game_code not in GAMES:
        print(f"\n\033[31m❌ Game '{game_code}' not configured\033[0m")
        return False

    game = GAMES[game_code]

    if not os.path.exists(game["rom"]):
        print(f"\n\033[31m❌ ROM not found:\033[0m")
        print(f"   {game['rom']}")
        return False

    if not os.path.exists(game["core"]):
        print(f"\n\033[31m❌ Core not found: {game['core']}\033[0m")
        return False

    cmd = RETROARCH_CMD + ["-L", game["core"], game["rom"]]
    playtime_sec = playtime_minutes * 60

    print(f"\n\033[36m🎮 Launching: {game['name']} ({game_code})\033[0m")
    print(f"\033[36m⏱️  Duration : {playtime_minutes} min\033[0m")
    print(f"\033[36m🧩 Core     : {os.path.basename(game['core'])}\033[0m")
    print(f"\033[36m📁 ROM      : {os.path.basename(game['rom'])}\033[0m")
    print("")

    try:
        proc = subprocess.Popen(
            cmd,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            stdin=subprocess.DEVNULL,
            preexec_fn=os.setsid,
        )
    except FileNotFoundError:
        print(f"\n\033[31m❌ Flatpak not found\033[0m")
        return False
    except Exception as e:
        print(f"\n\033[31m❌ Launch failed: {e}\033[0m")
        return False

    start_time = time.time()

    try:
        while True:
            elapsed = time.time() - start_time
            remaining = playtime_sec - elapsed

            if remaining <= 0:
                break

            if proc.poll() is not None:
                print("\n\033[33m⚠️  RetroArch closed early by user\033[0m")
                return True

            mins = int(remaining // 60)
            secs = int(remaining % 60)
            bar_width = 30
            filled = int((elapsed / playtime_sec) * bar_width)
            bar = "█" * filled + "░" * (bar_width - filled)

            sys.stdout.write(
                f"\r\033[35m⏳ [{bar}] \033[33m{mins:02d}:{secs:02d}\033[0m  "
            )
            sys.stdout.flush()
            time.sleep(1)

    except KeyboardInterrupt:
        print("\n\n\033[33m⚠️  Interrupted by operator\033[0m")

    print("\n\n\033[31m⏰ Time up! Closing game...\033[0m")
    kill_process_tree(proc)
    time.sleep(2)

    print("\033[32m✅ Session ended.\033[0m")
    return True


def kill_process_tree(proc):
    if proc.poll() is not None:
        return

    try:
        os.killpg(os.getpgid(proc.pid), signal.SIGTERM)
    except ProcessLookupError:
        return
    except Exception as e:
        print(f"  (SIGTERM failed: {e})")

    for _ in range(50):
        if proc.poll() is not None:
            return
        time.sleep(0.1)

    try:
        os.killpg(os.getpgid(proc.pid), signal.SIGKILL)
    except ProcessLookupError:
        pass
    except Exception as e:
        print(f"  (SIGKILL failed: {e})")
